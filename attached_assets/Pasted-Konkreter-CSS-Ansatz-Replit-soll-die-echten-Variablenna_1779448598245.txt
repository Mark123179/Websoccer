Abnahmebedingungen

Replit soll nach dem Fix diese Werte liefern:

(() => {
  const sidebar = document.querySelector(".navbar");
  const artboard =
    document.querySelector(".aaa-profile-artboard") ||
    document.querySelector(".player-dashboard");

  const season = document.querySelector(".aaa-season-card");
  const kpi = season?.querySelector(".aaa-kpis-scroll-wrap");

  const sidebarRect = sidebar?.getBoundingClientRect();
  const artboardRect = artboard?.getBoundingClientRect();

  console.table({
    viewport: `${window.innerWidth} × ${window.innerHeight}`,
    sidebarVisualRight: sidebarRect?.right,
    artboardVisualLeft: artboardRect?.left,
    artboardVisualRight: artboardRect?.right,
    viewportRight: window.innerWidth,
    artboardOffsetWidth: artboard?.offsetWidth,
    artboardClientWidth: artboard?.clientWidth,
    seasonClientWidth: season?.clientWidth,
    seasonScrollWidth: season?.scrollWidth,
    seasonDeltaX: season ? season.scrollWidth - season.clientWidth : null,
    kpiClientWidth: kpi?.clientWidth,
    kpiScrollWidth: kpi?.scrollWidth,
    kpiDeltaX: kpi ? kpi.scrollWidth - kpi.clientWidth : null,
    kpiOverflowX: kpi ? getComputedStyle(kpi).overflowX : null
  });
})();
