def minutes_adjusted_form(rating_form, played_minutes):
    if played_minutes == 0:
        return -2

    if played_minutes < 90:
        return min(0, round(rating_form * 0.25))

    if played_minutes < 270:
        return max(-3, min(2, round(rating_form * 0.50)))

    if played_minutes < 540:
        return max(-4, min(3, round(rating_form * 0.75)))

    return rating_form


def form_to_factor(form):
    mapping = {
        -5: 0.94,
        -4: 0.95,
        -3: 0.96,
        -2: 0.98,
        -1: 0.99,
         0: 1.00,
         1: 1.01,
         2: 1.02,
         3: 1.04,
         4: 1.05,
         5: 1.06,
    }

    return mapping.get(form, 1.00)
